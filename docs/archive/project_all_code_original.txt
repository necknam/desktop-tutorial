def _force_create_session(user_id=None, project_name=None, user_type="REAL_USER", *args, **kwargs):
    target_uid = user_id or kwargs.get("user_id") or "00000000-0000-0000-0000-000000000001"
    target_pname = project_name or kwargs.get("project_name") or kwargs.get("raw_user_prompt") or "?좉퇋 ?꾨줈?앺듃"
    if hasattr(db_client, "create_roadmap_session"):
        return db_client.create_roadmap_session(user_id=target_uid, project_name=str(target_pname)[:100])
    return str(uuid.uuid4())

        ok1 = db_client.save_roadmap_selection(session_id, generation_id) if hasattr(db_client,
                                                                                     "save_roadmap_selection") else True
        ok2 = db_client.save_roadmap_report(session_id, generation_id, detailed_report) if hasattr(db_client,
                                                                                                   "save_roadmap_report") else True
        return ok1 and ok2


    selection_service.select_roadmap = _safe_select

if not hasattr(selection_service, "get_selected_roadmap"):
    def _safe_get_selected(session_id):
        sel = db_client.get_roadmap_selection(session_id) if hasattr(db_client, "get_roadmap_selection") else None
        rep = db_client.get_roadmap_report(session_id) if hasattr(db_client, "get_roadmap_report") else None
        if not sel:
            return None
        gen_id = sel.get("generation_id")
        gens = db_client.get_roadmap_generations(session_id) if hasattr(db_client, "get_roadmap_generations") else []
        chosen_gen = next((g for g in gens if g.get("generation_id") == gen_id), {})
        return {"selection": sel, "generation": chosen_gen, "report": rep.get("report_content") if rep else {}}


    selection_service.get_selected_roadmap = _safe_get_selected

    def get_embedding(self, text: str) -> List[float]:
        """?⑥씪 ?띿뒪?몃? 1536李⑥썝 ?뺢퇋??踰≫꽣濡?蹂??""

            resp = self.client.embeddings.create(
                input=[clean_text],
                model=self.model_name
            )
            return resp.data[0].embedding
        except Exception as exc:
            logger.error(f"?꾨쿋??踰≫꽣 ?앹꽦 ?ㅽ뙣: {exc}")
            return [0.0] * 1536

    @staticmethod
    def cosine_similarity(
        vec_a: Union[List[float], np.ndarray], 
        vec_b: Union[List[float], np.ndarray]
    ) -> float:
        """??踰≫꽣 媛꾩쓽 肄붿궗???좎궗??(-1.0 ~ 1.0) ?곗궛"""
        a = np.array(vec_a, dtype=np.float32)
        b = np.array(vec_b, dtype=np.float32)
        norm_a = np.linalg.norm(a)
        norm_b = np.linalg.norm(b)

        if norm_a == 0.0 or norm_b == 0.0:
            return 0.0
        return float(np.dot(a, b) / (norm_a * norm_b))

class CircuitBreaker:
    """怨듦툒?먮퀎 ?낅┰ ?뚮줈 李⑤떒湲?""

    def __init__(self, provider_name: str, failure_threshold: int = 3, reset_timeout: float = 30.0) -> None:
        self.provider_name = provider_name
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.state = CircuitState.CLOSED
        self.consecutive_failures = 0
        self.last_failure_time = 0.0

    def can_execute(self) -> bool:
        """?꾩옱 ?몄텧 ?쒕룄 媛???щ? ?먮떒"""
        now = time.monotonic()
        if self.state == CircuitState.OPEN:
            if now - self.last_failure_time >= self.reset_timeout:
                logger.info(f"[{self.provider_name}] ?뚮줈 李⑤떒湲? OPEN -> HALF_OPEN (蹂듦뎄 ?먯깋 ?쒖옉)")
                self.state = CircuitState.HALF_OPEN
                return True
            return False
        return True

    def record_success(self) -> None:
        """?몄텧 ?깃났 ???곹깭 ?뺤긽??""
        if self.state != CircuitState.CLOSED:
            logger.info(f"[{self.provider_name}] ?뚮줈 李⑤떒湲? {self.state.value} -> CLOSED (?뺤긽 蹂듦뎄 ?꾨즺)")
        self.state = CircuitState.CLOSED
        self.consecutive_failures = 0

    def record_failure(self, error: Exception) -> None:
        """?몄텧 ?ㅽ뙣 ??移댁슫???꾩쟻 諛?李⑤떒 ?꾩씠"""
        self.consecutive_failures += 1
        self.last_failure_time = time.monotonic()
        logger.warning(
            f"[{self.provider_name}] ?몄텧 ?ㅽ뙣 ({self.consecutive_failures}/{self.failure_threshold}): {error}"
        )
        if self.consecutive_failures >= self.failure_threshold or self.state == CircuitState.HALF_OPEN:
            logger.error(
                f"[{self.provider_name}] ?뚮줈 李⑤떒湲?媛?? OPEN ?곹깭濡??꾩씠 ({self.reset_timeout}珥덇컙 李⑤떒)"
            )
            self.state = CircuitState.OPEN

            resp = client.post(url, json=payload, headers=headers)

    def generate(
        self,

        temperature: float = 0.5,
        force_provider: Optional[str] = None
    ) -> LLMResponse:
        """

        1?쒖쐞: OpenAI (gpt-4o-mini)
        2?쒖쐞: Local Qwen3.5 9B (Ollama / vLLM)

        """
        start_time = time.monotonic()
        errors: List[str] = []

        # ==================== 1?④퀎: OpenAI ?쒕룄 ====================
        if force_provider in (None, "openai"):
            breaker = self.breakers["openai"]
            if breaker.can_execute():

                    logger.info(f"OpenAI({self.openai_model}) ?몄텧 ?쒕룄...")
                    t0 = time.monotonic()

        "GENERIC_EMP_ID": re.compile(r"\b(?щ쾲\s*[:#]?\s*)([A-Za-z0-9_-]{4,12})\b"),

    def mask_text(self, text: str, custom_entities: Optional[List[str]] = None) -> str:
        """

        if custom_entities:

                if entity and len(entity.strip()) >= 2:

        # 2. ?뺢퇋?쒗쁽??湲곕컲 踰붿슜 PII 留덉뒪??

        masked = self.PATTERNS["TEL"].sub("[PHONE]", masked)
        masked = self.PATTERNS["EMP_CODE"].sub("[EMP_ID]", masked)
        masked = self.PATTERNS["GENERIC_EMP_ID"].sub(r"\1[EMP_ID]", masked)

        # 3. ?꾪삎?곸씤 ?쒓뎅???대쫫 ?몄묶 ?⑦꽩 留덉뒪??(?? ?띻만???곌뎄?? ?댁쿋???좎엫)

            r"[NAME] \2",

        )

        return masked

        """
        ?뺤뀛?덈━, 由ъ뒪????蹂듯빀 怨꾩링???곗씠???ш? 留덉뒪??

                    masked_dict[k] = "[EMAIL]"
                else:
                    masked_dict[k] = self.mask_data_structure(v, custom_entities)
            return masked_dict
        elif isinstance(data, list):
            return [self.mask_data_structure(item, custom_entities) for item in data]
        else:

def find_best_matching_resource(context_text: str, candidates: List[Dict[str, Any]], key_fields: List[str]) -> Dict[str, Any]:
    """留덉씪?ㅽ넠 臾몃㎘怨?RAG 寃???먯썝 媛꾩쓽 ?ㅼ썙???곹빀?꾨? 怨꾩궛?섏뿬 理쒖쟻 ?ㅼ〈 ?먯썝 1嫄?諛섑솚"""
    if not candidates:
        return {}

    ctx_lower = context_text.lower()
    best_item = candidates[0]
    best_score = -1

    for item in candidates:
        score = 0
        target_text = " ".join([str(item.get(f, "")) for f in key_fields]).lower()

        for kw in ["fastapi", "python", "postgresql", "sql", "docker", "kubernetes", "k8s", "redis", "aws", "azure", "ci/cd", "test", "security"]:
            if kw in ctx_lower and kw in target_text:
                score += 8

        words = [w for w in ctx_lower.split() if len(w) > 1]
        for w in words:
            if w in target_text:
                score += 1

        if score > best_score:
            best_score = score
            best_item = item

    return best_item

class MultiAgentOrchestrator:
    def __init__(self) -> None:
        self.agents = [
            PracticalProjectAgent(),
            CertifiedTheoryAgent(),
            FastTrackAgent(),
        ]

    def generate_all_roadmaps(
        self,
        user_context: Dict[str, Any],
        conversation_history: List[Dict[str, Any]],
        retrieved_context: Dict[str, Any],
        hidden_assessment_responses: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        t_start = time.monotonic()
        results: Dict[str, Any] = {}

        with ThreadPoolExecutor(max_workers=3) as executor:
            future_to_agent = {
                executor.submit(
                    agent.generate,
                    user_context,
                    conversation_history,
                    retrieved_context,
                    hidden_assessment_responses,
                ): agent.agent_id
                for agent in self.agents
            }

            for future in as_completed(future_to_agent):
                agent_id = future_to_agent[future]
                try:
                    res = future.result()
                    results[agent_id] = res
                except Exception as exc:
                    logger.error(f"?먯씠?꾪듃 [{agent_id}] ?ㅽ뻾 ?ㅽ뙣: {exc}")
                    results[agent_id] = {
                        "agent_id": agent_id,
                        "status": "FAILED",
                        "provider": "unknown",
                        "model": "unknown",
                        "latency_ms": int((time.monotonic() - t_start) * 1000),
                        "summary": f"?먯씠?꾪듃 ?ㅽ뻾 ?ㅽ뙣: {str(exc)}",
                        "roadmap_content": {"error": str(exc)},
                    }

        total_latency = int((time.monotonic() - t_start) * 1000)
        return {
            "total_latency_ms": total_latency,
            "results": results,
        }

    def fetch_dynamic_skills(self):
        """DB???ㅼ옱 ?곸옱?섏뼱 ?덈뒗 36媛??ㅽ궗怨??좏슚 釉뚮┸吏留??숈쟻 ?몄텧"""
        conn = self._get_connection()
        try:
            with conn.cursor(cursor_factory=psycopg2.extras.DictCursor) as cur:
                query = """
                        SELECT s.skill_id, \
                               s.name     AS skill_name, \
                               s.category AS skill_category, \

                               COALESCE( \
                                       json_agg(DISTINCT jsonb_build_object(
                                'id', j.job_id, 'title', j.title, 'ncs_units', j.ncs_units
                            )) FILTER(WHERE j.job_id IS NOT NULL), '[]' \
                               )          AS jobs, \

                               COALESCE( \
                                       json_agg(DISTINCT jsonb_build_object(
                                'id', c.course_id, 'title', c.title, 'platform', c.platform, 
                                'difficulty', c.difficulty, 'org', c.org_name, 'url', c.url, 
                                'duration_weeks', c.duration_weeks
                            )) FILTER(WHERE c.course_id IS NOT NULL), '[]' \
                               )          AS courses, \

                               COALESCE( \
                                       json_agg(DISTINCT jsonb_build_object(
                                'isbn', b.isbn, 'title', b.title, 'author', b.author, 'publisher', b.publisher
                            )) FILTER(WHERE b.isbn IS NOT NULL), '[]' \
                               )          AS books, \

                               COALESCE( \
                                       json_agg(DISTINCT jsonb_build_object(
                                'id', cr.cert_id, 'title', cr.title, 'provider', cr.provider, 
                                'pass_score', cr.pass_score, 'subjects', cr.test_subjects
                            )) FILTER(WHERE cr.cert_id IS NOT NULL), '[]' \
                               )          AS certs, \

                               COALESCE( \
                                       json_agg(DISTINCT jsonb_build_object(
                                'id', m.major_id, 'title', m.title, 'curriculum', m.curriculum
                            )) FILTER(WHERE m.major_id IS NOT NULL), '[]' \
                               )          AS majors

                        FROM skill_tags s
                                 LEFT JOIN job_skill_map jm ON s.skill_id = jm.skill_id
                                 LEFT JOIN jobs j ON jm.job_id = j.job_id
                                 LEFT JOIN course_skill_map cm ON s.skill_id = cm.skill_id
                                 LEFT JOIN courses c ON cm.course_id = c.course_id
                                 LEFT JOIN book_skill_map bm ON s.skill_id = bm.skill_id
                                 LEFT JOIN books b ON bm.isbn = b.isbn
                                 LEFT JOIN cert_skill_map crm ON s.skill_id = crm.skill_id
                                 LEFT JOIN certifications cr ON crm.cert_id = cr.cert_id
                                 LEFT JOIN major_skill_map mm ON s.skill_id = mm.skill_id
                                 LEFT JOIN majors m ON mm.major_id = m.major_id
                        GROUP BY s.skill_id, s.name, s.category
                        ORDER BY s.skill_id; \
                        """
                cur.execute(query)
                return [dict(r) for r in cur.fetchall()]
        finally:
            conn.close()
